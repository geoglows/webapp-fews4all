% Engine-independent setup. Tectonic runs XeTeX, which wants fontspec and has no
% use for inputenc; a full TeX Live install running pdflatex wants the opposite.
% Branching here means the same file builds under either without being edited.

\ifPDFTeX
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
  \usepackage{lmodern}
\else

\fi
